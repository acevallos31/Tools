function Get-RedirectStatus {

    param(
        [string]$StartUrl,
        [int]$MaxHops = 10
    )

    $url = $StartUrl
    $resultado = @()

    for($i=1; $i -le $MaxHops; $i++){

        try{

            $req = [System.Net.HttpWebRequest]::Create($url)
            $req.Method = "GET"
            $req.AllowAutoRedirect = $false
            $req.Timeout = 10000
            $req.UserAgent = "EndpointCentral-Diag"

            $resp = $req.GetResponse()

            $code = [int]$resp.StatusCode
            $location = $resp.Headers["Location"]

            $resultado += "[$i] $code $url"

            if($location){

                if($location -match '^https?://'){
                    $url = $location
                }
                else{
                    $base = [uri]$url
                    $url = [uri]::new($base,$location).AbsoluteUri
                }

                $resp.Close()
                continue
            }

            $resp.Close()
            break

        }
        catch{

            if($_.Exception.Response){

                $resp = $_.Exception.Response
                $code = [int]$resp.StatusCode
                $location = $resp.Headers["Location"]

                $resultado += "[$i] $code $url"

                if($location){

                    if($location -match '^https?://'){
                        $url = $location
                    }
                    else{
                        $base = [uri]$url
                        $url = [uri]::new($base,$location).AbsoluteUri
                    }

                    $resp.Close()
                    continue
                }

                $resp.Close()
                break
            }
            else{

                $resultado += "[$i] ERROR $url -> $($_.Exception.Message)"
                break
            }
        }
    }

    return $resultado
}

for($i=1;$i -le 20;$i++){

    $tcp=$false
    $client=New-Object Net.Sockets.TcpClient

    try{

        $async=$client.BeginConnect($IP,443,$null,$null)

        if($async.AsyncWaitHandle.WaitOne(3000)){
            $client.EndConnect($async)
            $tcp=$true
        }

    }catch{}

    $client.Close()

    $issuer="-"
    $http="-"

    if($tcp){

        try{

            $tc=New-Object Net.Sockets.TcpClient($H,443)

            $ssl=New-Object Net.Security.SslStream(
                $tc.GetStream(),
                $false,
                ({$true} -as [Net.Security.RemoteCertificateValidationCallback])
            )

            $ssl.AuthenticateAsClient($H)

            $issuer=$ssl.RemoteCertificate.Issuer

            $ssl.Close()
            $tc.Close()

        }catch{

            $issuer="TLS-ERR"
        }

        $redirects = Get-RedirectStatus "https://$H"

        $http = $redirects -join " | "
    }

    L ("{0}  {1,-6}  {2,-35}  {3}" -f `
        (Get-Date -Format 'HH:mm:ss'),
        $tcp,
        ($issuer -replace '^CN=',''),
        $http
    )

    if($i -lt 20){
        Start-Sleep -Seconds 30
    }
}
